async function syncPendingAssessments() { if (!navigator.onLine) return; let queue = []; try { queue = JSON.parse(localStorage.getItem('alpha_pending_assessments') || '[]'); } catch (e) { return; } if (queue.length === 0) return; console.log(`Starting sync of ${queue.length} pending assessments...`); let remaining = []; let syncedCount = 0; for (let item of queue) { try { let formData = new FormData(); for (let key in item) { if (key !== 'saved_at') { // Rebuild the arrays correctly for multiple hazards if (Array.isArray(item[key])) { item[key].forEach(val => formData.append(key, val)); } else { formData.append(key, item[key]); } } } let response = await fetch('/new', { method: 'POST', body: formData }); // If the user logged out, save it back to the queue and stop. if (response.redirected && response.url.includes('/login')) { console.log('Session expired. Pausing sync.'); remaining.push(item); continue; } if (response.ok) { syncedCount++; } else { remaining.push(item); } } catch (err) { console.error('Sync failed for item:', err); remaining.push(item); } } localStorage.setItem('alpha_pending_assessments', JSON.stringify(remaining)); if (syncedCount > 0) { console.log(`Successfully synced ${syncedCount} assessments.`); if (window.location.pathname === '/dashboard' || window.location.pathname === '/history') { location.reload(); } } }
